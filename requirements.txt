# ---------------------------------------------------------------------------
# RECOMMENDED for local serving: the GGUF engine only.
#
#   macOS (Metal):  CMAKE_ARGS="-DGGML_METAL=on" pip install llama-cpp-python
#   Linux/CPU:      pip install llama-cpp-python
#
# That single package is enough to serve a quantised policy — you do NOT need
# torch/transformers/peft below unless you want to run the unquantised model or
# attach LoRA adapters directly. A 4B Q4_K_M GGUF needs ~2.5GB; the same policy
# in fp16 through transformers needs ~8GB.
#
# Use an arm64 Python on Apple Silicon: an x86 build runs ~10x slower.
# ---------------------------------------------------------------------------
llama-cpp-python>=0.3.2

#   CUDA: pip install --index-url https://download.pytorch.org/whl/cu129 torch==2.13.0
#   macOS (MPS): pip install torch==2.13.0
